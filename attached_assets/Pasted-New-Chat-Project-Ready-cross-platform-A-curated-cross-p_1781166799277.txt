WITH paid_orders AS (
    SELECT customer_id, total
    FROM orders
    WHERE status = 'paid'
)
SELECT c.name,
       COUNT(paid_orders.customer_id) AS paid_order_count,
       COALESCE(SUM(paid_orders.total), 0) AS paid_revenue
FROM customers c
LEFT JOIN paid_orders ON paid_orders.customer_id = c.id
GROUP BY c.id, c.name
ORDER BY paid_revenue DESC
LIMIT 8;
